cmake_minimum_required(VERSION 3.20)
project(tsfp_mod_worker C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT EXISTS "${TSFP_LUA_SOURCE}/lua.h")
    message(FATAL_ERROR "Use tools/build/build-mod-worker.py to verify/prepare pinned Lua")
endif()
# No CLI, package loader, debug, IO, OS, coroutine or open-all library.
set(lua_core lapi.c lcode.c lctype.c ldebug.c ldo.c ldump.c lfunc.c lgc.c llex.c
    lmem.c lobject.c lopcodes.c lparser.c lstate.c lstring.c ltable.c ltm.c
    lundump.c lvm.c lzio.c lauxlib.c lbaselib.c lmathlib.c lstrlib.c ltablib.c lutf8lib.c)
list(TRANSFORM lua_core PREPEND "${TSFP_LUA_SOURCE}/")
add_library(tsfp_mod_vm STATIC ${lua_core} mod_script.c)
target_include_directories(tsfp_mod_vm PRIVATE "${TSFP_LUA_SOURCE}")
target_include_directories(tsfp_mod_vm PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(tsfp_mod_vm PRIVATE _CRT_SECURE_NO_WARNINGS)
add_executable(tsfp-mod-worker mod_worker.c mod_watchdog.c mod_seal.c)
target_link_libraries(tsfp-mod-worker PRIVATE tsfp_mod_vm)
if(WIN32)
    target_link_libraries(tsfp-mod-worker PRIVATE advapi32)
endif()
if(TSFP_MOD_TESTING)
    add_executable(tsfp-mod-process-probe ../../tests/mods/native_process_probe.c mod_seal.c)
    target_include_directories(tsfp-mod-process-probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    if(WIN32)
        target_link_libraries(tsfp-mod-process-probe PRIVATE advapi32 ws2_32 userenv)
        set_property(TARGET tsfp-mod-process-probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_executable(tsfp-mod-vm-test mod_worker.c mod_watchdog.c)
    target_compile_definitions(tsfp-mod-vm-test PRIVATE TSFP_MOD_VM_TEST_ONLY)
    target_link_libraries(tsfp-mod-vm-test PRIVATE tsfp_mod_vm)
    if(NOT WIN32)
        find_package(Threads REQUIRED)
        target_link_libraries(tsfp-mod-vm-test PRIVATE m Threads::Threads)
    endif()
endif()
if(WIN32)
    add_library(tsfp_mod_process STATIC mod_process_windows.cpp)
    target_link_libraries(tsfp_mod_process PUBLIC userenv advapi32 psapi ole32)
else()
    add_library(tsfp_mod_process STATIC mod_process_posix.cpp)
endif()
target_include_directories(tsfp_mod_process PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(tsfp-mod-runner mod_runner.cpp)
target_link_libraries(tsfp-mod-runner PRIVATE tsfp_mod_process)
if(TSFP_MOD_TESTING)
    add_executable(tsfp-mod-lifetime-probe ../../tests/mods/native_lifetime_probe.cpp)
    target_link_libraries(tsfp-mod-lifetime-probe PRIVATE tsfp_mod_process)
    if(MSVC)
        set_property(TARGET tsfp-mod-lifetime-probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endif()
if(APPLE)
    add_executable(tsfp-mod-guardian mod_guardian_mac.c)
    target_compile_options(tsfp-mod-guardian PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
endif()
if(NOT WIN32)
    find_package(Threads REQUIRED)
    target_link_libraries(tsfp-mod-worker PRIVATE m Threads::Threads)
    target_link_libraries(tsfp_mod_process PUBLIC Threads::Threads)
    target_link_libraries(tsfp-mod-runner PRIVATE Threads::Threads)
endif()
if(MSVC)
    foreach(target tsfp_mod_vm tsfp-mod-worker tsfp_mod_process tsfp-mod-runner)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endforeach()
    if(TSFP_MOD_TESTING)
        set_property(TARGET tsfp-mod-vm-test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    target_compile_options(tsfp_mod_vm PRIVATE /W4 /guard:cf)
    target_compile_options(tsfp-mod-worker PRIVATE /W4 /guard:cf)
    target_link_options(tsfp-mod-worker PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
else()
    target_compile_options(tsfp_mod_vm PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
    target_compile_options(tsfp-mod-worker PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
    if(TSFP_MOD_SANITIZE)
        foreach(target tsfp_mod_vm tsfp-mod-worker)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endforeach()
        if(TSFP_MOD_TESTING)
            target_compile_options(tsfp-mod-vm-test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(tsfp-mod-vm-test PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endif()

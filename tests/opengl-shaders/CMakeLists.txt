cmake_minimum_required(VERSION 3.20)
project(tsfp_opengl_shader_contract C CXX)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(runtime "${root}/external/ts3FPrecomp")
add_executable(tsfp_opengl_shader_test shaders.cpp "${root}/src/graphics/opengl_shader.cpp"
    "${runtime}/src/d3d/rhi_vulkan_dxc.cpp")
target_include_directories(tsfp_opengl_shader_test PRIVATE "${root}/src/graphics" "${runtime}/src/d3d")
set_target_properties(tsfp_opengl_shader_test PROPERTIES CXX_STANDARD 17)
include("${root}/src/graphics/spirv_cross.cmake")
tsfp_add_spirv_cross(tsfp_opengl_shader_test)
add_executable(tsfp_opengl_pixel_test pixels.cpp "${root}/src/graphics/opengl_host.c")
target_include_directories(tsfp_opengl_pixel_test PRIVATE "${root}/src/graphics"
    "${runtime}/third_party/SDL/include")
set_target_properties(tsfp_opengl_pixel_test PROPERTIES CXX_STANDARD 17 C_STANDARD 11)
if(WIN32)
    target_link_libraries(tsfp_opengl_pixel_test PRIVATE opengl32 user32 gdi32)
    set(dxc "${runtime}/third_party/dxc")
    target_compile_definitions(tsfp_opengl_shader_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_include_directories(tsfp_opengl_shader_test PRIVATE "${dxc}/inc")
    foreach(dll dxcompiler dxil)
        add_custom_command(TARGET tsfp_opengl_shader_test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${dxc}/bin/x64/${dll}.dll" "$<TARGET_FILE_DIR:tsfp_opengl_shader_test>")
    endforeach()
else()
    find_package(SDL3 CONFIG REQUIRED)
    target_link_libraries(tsfp_opengl_pixel_test PRIVATE SDL3::SDL3)
    if(APPLE)
        find_library(opengl OpenGL REQUIRED)
        target_link_libraries(tsfp_opengl_pixel_test PRIVATE "${opengl}")
    else()
        find_package(OpenGL REQUIRED)
        target_link_libraries(tsfp_opengl_pixel_test PRIVATE OpenGL::GL)
    endif()
    if(NOT TSFP_VULKAN_SDK)
        message(FATAL_ERROR "Pass the verified host SDK with -DTSFP_VULKAN_SDK=...")
    endif()
    target_include_directories(tsfp_opengl_shader_test PRIVATE "${TSFP_VULKAN_SDK}/include/dxc")
    if(APPLE)
        set(dxc "${TSFP_VULKAN_SDK}/lib/libdxcompiler.dylib")
    else()
        set(dxc "${TSFP_VULKAN_SDK}/lib/libdxcompiler.so")
    endif()
    target_compile_definitions(tsfp_opengl_shader_test PRIVATE "RHI_DXC_LIBRARY=\"${dxc}\"")
    target_link_libraries(tsfp_opengl_shader_test PRIVATE ${CMAKE_DL_LIBS})
endif()
